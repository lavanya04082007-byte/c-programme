#include <stdio.h>
#define MAX 10

int queue[MAX];
int front = -1, rear = -1;

int isEmpty() {
    return (front == -1 || front > rear);
}

void enqueue(int jobID) {
    if (rear == MAX - 1) {
        printf("Printer queue is full\n");
    } else {
        if (front == -1)
            front = 0;
        queue[++rear] = jobID;
        printf("Job added: %d\n", jobID);
    }
}

void dequeue() {
    if (isEmpty()) {
        printf("No print jobs pending\n");
    } else {
        printf("Job printed: %d\n", queue[front++]);
    }
}

void showFront() {
    if (isEmpty()) {
        printf("No next job\n");
    } else {
        printf("Next job: %d\n", queue[front]);
    }
}

void size() {
    if (isEmpty())
        printf("Pending jobs: 0\n");
    else
        printf("Pending jobs: %d\n", rear - front + 1);
}

int main() {
    enqueue(1);   
    enqueue(2);   
    enqueue(3);   

    dequeue();    
    showFront();  
    size();       

    return 0;
}
