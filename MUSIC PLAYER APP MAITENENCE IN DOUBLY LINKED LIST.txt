#include <stdio.h>

#include <stdlib.h>


struct Node {
  
  int stationID;

    struct Node *prev;

    struct Node *next;

};


struct Node* insertAtEnd(struct Node *head, int data) {
 
   struct Node *newNode = (struct Node*)malloc(sizeof(struct Node));

    newNode->stationID = data;

    newNode->next = NULL;


    if (head == NULL) {

        newNode->prev = NULL;

        return newNode;

    }


    struct Node *temp = head;

    while (temp->next != NULL)

        temp = temp->next;


    temp->next = newNode;

    newNode->prev = temp;


    return head;

}


void display(struct Node *head) {
 
   struct Node *temp = head;

    while (temp != NULL) {

        printf("%d<>", temp->stationID);

        temp = temp->next;

    }

    printf("NULL");

}


int main() {

    struct Node *head = NULL;


    head = insertAtEnd(head, 101);

    head = insertAtEnd(head, 102);

    head = insertAtEnd(head, 103);


    display(head);

    return 0;

}
