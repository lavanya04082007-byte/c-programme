#include <stdio.h>

int main() {
    int choice;

    printf("Welcome to Telecaller Service\n");
    printf("Press 1 for Tamil\n");
    printf("Press 2 for English\n");
    printf("Press 3 for Hindi\n");
    printf("Enter your choice: ");
    scanf("%d", &choice);

    switch(choice) {
        case 1:
            printf("You have selected Tamil.\n");
            break;
        case 2:
            printf("You have selected English.\n");
            break;
        case 3:
            printf("You have selected Hindi.\n");
            break;
        default:
            printf("Invalid choice. Please try again.\n");
    }

    return 0;
}
